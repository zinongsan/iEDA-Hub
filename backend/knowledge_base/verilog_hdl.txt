# Verilog HDL编程详解

## Verilog简介

**Verilog是什么**：Verilog是一种硬件描述语言（HDL），用于描述数字电路和系统的行为和结构。它不是编程语言，而是用文本方式描述硬件电路的语言。

**Verilog的作用**：使用Verilog，设计师可以用类似C语言的语法来描述复杂的数字电路，然后由EDA工具自动转换为实际的硬件电路。它广泛应用于ASIC和FPGA设计。

**为什么学习Verilog**：
- 数字IC设计的必备技能
- 比画原理图更高效
- 便于复杂电路的设计和验证
- 行业标准，就业必备
- 支持仿真和综合

## Verilog的抽象层次

### 1. 行为级（Behavioral Level）
使用高级语句描述电路功能，不关心具体实现。

```verilog
// 行为级：8位加法器
module adder8(
    input [7:0] a, b,
    output reg [7:0] sum
);
    always @(*) begin
        sum = a + b;
    end
endmodule
```

### 2. 寄存器传输级（RTL Level）
描述数据在寄存器之间的传输和处理，这是综合工具的输入层次。

```verilog
// RTL级：带使能的8位寄存器
module reg8(
    input clk,
    input rst_n,
    input en,
    input [7:0] d,
    output reg [7:0] q
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            q <= 8'b0;
        else if (en)
            q <= d;
    end
endmodule
```

### 3. 门级（Gate Level）
使用基本逻辑门描述电路，通常是综合工具的输出。

```verilog
// 门级：全加器
module full_adder(
    input a, b, cin,
    output sum, cout
);
    wire s1, c1, c2;

    xor (s1, a, b);
    xor (sum, s1, cin);
    and (c1, a, b);
    and (c2, s1, cin);
    or (cout, c1, c2);
endmodule
```

## Verilog基本语法

### 模块定义
```verilog
module module_name (
    // 端口声明
    input wire [7:0] data_in,
    input wire clk, rst_n,
    output reg [7:0] data_out,
    output wire valid
);
    // 内部信号声明
    reg [3:0] counter;
    wire [7:0] temp;

    // 逻辑描述
    // ...

endmodule
```

### 数据类型

#### 线网类型（Net Types）
- **wire**：连接门和模块，不能存储值
- **tri**：三态线网
- **supply0/supply1**：电源和地

#### 寄存器类型（Register Types）
- **reg**：可以存储值，用于过程赋值
- **integer**：整数类型，32位有符号
- **real**：实数类型，用于仿真
- **time**：时间类型，64位无符号

### 数值表示
```verilog
// 格式：<位宽>'<进制><数值>
8'b1010_1010    // 8位二进制

8'd170          // 8位十进制
4'b10x1         // x表示未知态
4'b10z1         // z表示高阻态
```

## 组合逻辑设计

### always块 - 组合逻辑
```verilog
// 使用 always @(*) 或 always_comb
always @(*) begin
    // 组合逻辑
    case (sel)
        2'b00: out = a;
        2'b01: out = b;
        2'b10: out = c;
        2'b11: out = d;
    endcase
end
```

### assign语句
```verilog
// 连续赋值
assign sum = a + b;
assign carry = (a + b) > 15;
```

### 条件运算符
```verilog
assign out = sel ? a : b;  // 2选1多路器
assign out = (sel == 2'b00) ? a :
             (sel == 2'b01) ? b :
             (sel == 2'b10) ? c : d;
```

## 时序逻辑设计

### 同步复位D触发器
```verilog
always @(posedge clk) begin
    if (rst) begin
        q <= 1'b0;
    end else begin
        q <= d;
    end
end
```

### 异步复位D触发器
```verilog
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        q <= 1'b0;
    end else begin
        q <= d;
    end
end
```

### 使能信号
```verilog
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        counter <= 8'b0;
    end else if (en) begin
        counter <= counter + 1'b1;
    end
end
```

## 状态机设计

### 两段式状态机
```verilog
// 状态定义
localparam IDLE  = 2'b00;
localparam LOAD  = 2'b01;
localparam PROC  = 2'b10;
localparam DONE  = 2'b11;

reg [1:0] current_state, next_state;

// 状态转移（时序逻辑）
always @(posedge clk or negedge rst_n) begin
    if (!rst_n)
        current_state <= IDLE;
    else
        current_state <= next_state;
end

// 下一状态逻辑（组合逻辑）
always @(*) begin
    case (current_state)
        IDLE: begin
            if (start)
                next_state = LOAD;
            else
                next_state = IDLE;
        end
        LOAD: begin
            next_state = PROC;
        end
        PROC: begin
            if (done)
                next_state = DONE;
            else
                next_state = PROC;
        end
        DONE: begin
            next_state = IDLE;
        end
        default: next_state = IDLE;
    endcase
end

// 输出逻辑（组合逻辑）
always @(*) begin
    case (current_state)
        IDLE: begin
            busy = 1'b0;
            valid = 1'b0;
        end
        LOAD: begin
            busy = 1'b1;
            valid = 1'b0;
        end
        PROC: begin
            busy = 1'b1;
            valid = 1'b0;
        end
        DONE: begin
            busy = 1'b0;
            valid = 1'b1;
        end
    endcase
end
```

### 三段式状态机
```verilog
// 第三段：输出逻辑（时序）
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        busy <= 1'b0;
        valid <= 1'b0;
    end else begin
        case (current_state)
            IDLE: begin
                busy <= 1'b0;
                valid <= 1'b0;
            end
            LOAD: begin
                busy <= 1'b1;
                valid <= 1'b0;
            end
            // ...
        endcase
    end
end
```

## 常见设计模式

### 计数器
```verilog
module counter #(
    parameter WIDTH = 8
)(
    input clk,
    input rst_n,
    input en,
    input load,
    input [WIDTH-1:0] load_value,
    output reg [WIDTH-1:0] count
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            count <= {WIDTH{1'b0}};
        else if (load)
            count <= load_value;
        else if (en)
            count <= count + 1'b1;
    end
endmodule
```

### FIFO
```verilog
module fifo #(
    parameter DATA_WIDTH = 8,
    parameter DEPTH = 16,
    parameter ADDR_WIDTH = 4
)(
    input clk,
    input rst_n,
    input wr_en,
    input [DATA_WIDTH-1:0] wr_data,
    input rd_en,
    output reg [DATA_WIDTH-1:0] rd_data,
    output full,
    output empty
);
    reg [DATA_WIDTH-1:0] mem [0:DEPTH-1];
    reg [ADDR_WIDTH:0] wr_ptr, rd_ptr;

    assign full = (wr_ptr[ADDR_WIDTH] != rd_ptr[ADDR_WIDTH]) &&
                  (wr_ptr[ADDR_WIDTH-1:0] == rd_ptr[ADDR_WIDTH-1:0]);
    assign empty = (wr_ptr == rd_ptr);

    // 写操作
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            wr_ptr <= {(ADDR_WIDTH+1){1'b0}};
        else if (wr_en && !full) begin
            mem[wr_ptr[ADDR_WIDTH-1:0]] <= wr_data;
            wr_ptr <= wr_ptr + 1'b1;
        end
    end

    // 读操作
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            rd_ptr <= {(ADDR_WIDTH+1){1'b0}};
        else if (rd_en && !empty) begin
            rd_data <= mem[rd_ptr[ADDR_WIDTH-1:0]];
            rd_ptr <= rd_ptr + 1'b1;
        end
    end
endmodule
```

### 握手协议
```verilog
// Valid-Ready握手
module handshake_example (
    input clk,
    input rst_n,

    // 输入接口
    input in_valid,
    input [7:0] in_data,
    output reg in_ready,

    // 输出接口
    output reg out_valid,
    output reg [7:0] out_data,
    input out_ready
);
    // 数据处理逻辑
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            out_valid <= 1'b0;
            out_data <= 8'b0;
            in_ready <= 1'b1;
        end else begin
            if (in_valid && in_ready) begin
                // 接收数据
                out_data <= in_data + 8'd1;  // 简单处理
                out_valid <= 1'b1;
                in_ready <= 1'b0;
            end

            if (out_valid && out_ready) begin
                // 数据被接收
                out_valid <= 1'b0;
                in_ready <= 1'b1;
            end
        end
    end
endmodule
```

## 可综合编码规范

### ✅ 推荐写法

1. **使用非阻塞赋值（<=）在时序逻辑中**
```verilog
always @(posedge clk) begin
    q <= d;  // 正确
end
```

2. **使用阻塞赋值（=）在组合逻辑中**
```verilog
always @(*) begin
    temp = a + b;
    out = temp * c;  // 正确
end
```

3. **完整的敏感列表**
```verilog
always @(*) begin  // 使用 @(*) 自动推导
    // 组合逻辑
end
```

4. **完整的if-else和case**
```verilog
always @(*) begin
    case (sel)
        2'b00: out = a;
        2'b01: out = b;
        2'b10: out = c;
        default: out = d;  // 必须有default
    endcase
end
```

### ❌ 不推荐写法

1. **混用阻塞和非阻塞赋值**
```verilog
always @(posedge clk) begin
    q = d;  // 错误：时序逻辑应该用 <=
end
```

2. **不完整的敏感列表**
```verilog
always @(a) begin  // 错误：缺少b
    out = a + b;
end
```

3. **锁存器推断（无意的）**
```verilog
always @(*) begin
    if (sel)
        out = a;
    // 缺少else，会推断出锁存器
end
```

4. **使用延迟语句**
```verilog
always @(posedge clk) begin
    #10 q = d;  // 不可综合
end
```

module sync_2ff (

    input async_in,
    output sync_out

## 测试台（Testbench）编写

```verilog
`timescale 1ns/1ps

module tb_counter;
    // 信号声明
    reg clk;
    reg rst_n;
    reg en;
    wire [7:0] count;

    // 实例化待测模块
    counter #(.WIDTH(8)) dut (
        .clk(clk),
        .rst_n(rst_n),
        .en(en),
        .load(1'b0),
        .load_value(8'b0),
        .count(count)
    );

    // 时钟生成
    initial begin
        clk = 0;
        forever #5 clk = ~clk;  // 100MHz
    end

    // 复位和激励
    initial begin
        rst_n = 0;
        en = 0;
        #20 rst_n = 1;
        #10 en = 1;
        #200 en = 0;
        #50 en = 1;
        #100 $finish;
    end

    // 监控输出
    initial begin
        $monitor("Time=%0t rst_n=%b en=%b count=%d",
                 $time, rst_n, en, count);
    end

    // 波形导出
    initial begin
        $dumpfile("counter.vcd");
        $dumpvars(0, tb_counter);
    end
endmodule
```

## 常见错误和调试技巧

### 1. 组合逻辑环路
```verilog
// 错误示例
assign a = b;
assign b = c;
assign c = a;  // 形成环路
```

### 2. 多驱动
```verilog
// 错误示例
assign out = a;
assign out = b;  // out被多次驱动
```

### 3. 未初始化的寄存器
```verilog
// 仿真可能工作，但综合后不确定
reg [7:0] data;  // 没有复位初始化
```

### 4. 时序路径过长
```verilog
// 一个时钟周期内做太多事
always @(posedge clk) begin
    result = ((a + b) * c) - ((d + e) * f);  // 太复杂
end
```

## SystemVerilog增强特性

### 逻辑类型
```systemverilog
logic [7:0] data;  // 可替代wire和reg
```

### always_comb / always_ff
```systemverilog
always_comb begin  // 组合逻辑
    sum = a + b;
end

always_ff @(posedge clk) begin  // 时序逻辑
    q <= d;
end
```

### 接口（Interface）
```systemverilog
interface axi_if;
    logic valid;
    logic ready;
    logic [31:0] data;
endinterface
```
